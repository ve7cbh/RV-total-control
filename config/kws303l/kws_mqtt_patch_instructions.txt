# Patch for kws_mqtt.py — add relay-state read-back
#
# This adds register 63 (the relay state) to the existing poll list, so
# you get a real sensor.grid_relay_state / sensor.generator_relay_state
# reflecting actual hardware state — no input_boolean guessing needed,
# unlike the Waveshare coils.
#
# HOW TO APPLY:
# Open kws_mqtt.py and find the KWS_REGISTERS list. Add this one line
# to it (anywhere in the list, order doesn't matter):

    (63,       1.0,    "relay_state",       ""),      # 1=connected, 0=disconnected (confirmed via mbpoll 2026-06-30)

# So the block should look like:
#
# KWS_REGISTERS = [
#     # address  scale    field               unit
#     (14,       0.01,   "voltage",           "V"),
#     (18,       0.001,  "current",           "A"),
#     (26,       0.1,    "power",             "W"),
#     (48,       0.001,  "power_factor",      ""),
#     (51,       0.01,   "frequency",         "Hz"),
#     (55,       0.001,  "energy_kwh",        "kWh"),
#     (60,       1.0,    "temperature",       "C"),
#     (62,       1.0,    "alarm_code",        ""),
#     (63,       1.0,    "relay_state",       ""),
# ]
#
# No other changes needed — poll_meter() already loops over this list
# generically and will publish rvtc/sensors/grid/relay_state (and
# rvtc/sensors/generator/relay_state once GENERATOR_ENABLED = True)
# automatically.
#
# After editing, restart the kws_mqtt service:
#   sudo systemctl restart kws_mqtt.service
#
# Then confirm in HA Developer Tools -> States that a new sensor shows
# up reading the relay state — likely named something like
# sensor.grid_relay_state if your MQTT sensor auto-discovery follows
# the same pattern as the other grid/* topics already in HA. If it
# doesn't appear automatically, check mqtt_sensors.yaml — you may need
# to add an explicit MQTT sensor entry there the same way the existing
# grid sensors were defined, since this is a new field on an existing
# topic tree rather than a new topic tree.
